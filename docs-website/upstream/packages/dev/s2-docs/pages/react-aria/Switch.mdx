## Selection

Use the `isSelected` or `defaultSelected` prop to set the selection state, and `onChange` to handle selection changes.

```tsx render solid="./Switch/solid-selection.tsx"
"use client";
import {Switch} from './Switch/vanilla/Switch.solid';
import {createSignal} from 'solid-js';

function Example(props) {
  let [selected, setSelection] = createSignal(false);

  return (
    <>
      <Switch
        {...props}
        ///- begin highlight -///
        isSelected={selected()}
        onChange={setSelection}
        ///- end highlight -///
      >
        Low power mode
      </Switch>
      <p>{selected() ? 'Low' : 'High'} power mode active.</p>
    </>
  );
}
```

## Forms

Use the `name` and `value` props to submit the switch to the server. Set the `isRequired` prop to validate the user selects the switch, or implement custom client or server-side validation. See the [Forms](forms) guide to learn more.

```tsx render solid="./Switch/solid-forms.tsx"
"use client";
import {Switch} from './Switch/vanilla/Switch.solid';
import {Button} from './Button/vanilla/Button.solid';
import {Form} from './Button/vanilla/Form.solid';

<Form>
  {/*- begin highlight -*/}
  <Switch
    name="two-factor"
    isRequired
    description="Your organization requires two-factor authentication.">
    Two-factor authentication
  </Switch>
  {/*- end highlight -*/}
  <Button type="submit" style={{'margin-top': '8px'}}>Submit</Button>
</Form>
```
