## Selection

Use the `isSelected` or `defaultSelected` prop to set the selection state, and `onChange` to handle selection changes.

```tsx render solid="./ToggleButton/solid-selection.tsx"
"use client";
import {createSignal} from 'solid-js';
import {ToggleButton} from './ToggleButton/vanilla/ToggleButton.solid';
import {Star} from './Button/icons.solid';

function Example(props) {
  let [selected, setSelection] = createSignal(false);

  return (
    <>
      <ToggleButton
        {...props}
        aria-label="Star"
        /*- begin highlight -*/
        isSelected={selected()}
        onChange={setSelection}>
        {/*- end highlight -*/}
        <Star size={18} />
      </ToggleButton>
      <p>{selected() ? 'Starred' : 'Not starred'}</p>
    </>
  );
}
```
