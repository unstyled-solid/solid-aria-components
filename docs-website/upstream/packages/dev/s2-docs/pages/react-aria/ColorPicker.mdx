## Value

Use the `value` or `defaultValue` prop to set the color value. This may be a string or <TypeLink type={docs.exports.Color} links={docs.links} /> object, parsed using the <TypeLink type={docs.exports.parseColor} links={docs.links} /> function. The `onChange` event is always called with a `Color` object.

```tsx nativeModule="ColorPicker/solid-example-1.tsx" solid="ColorPicker/solid-example-1.tsx" files={["packages/dev/s2-docs/pages/react-aria/ColorPicker/solid-example-1.tsx"]} 
import { parseColor } from 'solid-aria-components';
import { ColorPicker } from './ColorPicker/vanilla/ColorPicker.solid';
import { createSignal } from 'solid-js';
function Example() {
    let [value, setValue] = createSignal(parseColor('hsl(50, 100%, 50%)'));
    return (<>
      <ColorPicker label="Color" 
    /*- begin highlight -*/
    value={value()} onChange={setValue}/>
        {/*- end highlight -*/}
      <pre style={{ "font-size": "12px" }}>Selected color: {value().toString('hsl')}</pre>
    </>);
}
```

## Examples

### Channel sliders

This example uses [ColorSlider](ColorSlider) to allow a user to adjust each channel of a color value, with a [Select](Select) to switch between color spaces.

```tsx nativeModule="ColorPicker/solid-example-2.tsx" solid="ColorPicker/solid-example-2.tsx" files={["packages/dev/s2-docs/pages/react-aria/ColorPicker/solid-example-2.tsx"]} 
import { getColorChannels, type ColorSpace } from 'solid-aria-components';
import { ColorPicker } from './ColorPicker/vanilla/ColorPicker.solid';
import { ColorSlider } from './ColorPicker/vanilla/ColorSlider.solid';
import { Select, SelectItem } from './ColorPicker/vanilla/Select.solid';
import { createSignal, For } from 'solid-js';
function Example() {
    let [space, setSpace] = createSignal<ColorSpace>('rgb');
    return (<ColorPicker label="Fill color" defaultValue="#184">
      <Select aria-label="Color space" selectedKey={space()} onSelectionChange={s => setSpace(s as ColorSpace)}>
        <SelectItem id="rgb">RGB</SelectItem>
        <SelectItem id="hsl">HSL</SelectItem>
        <SelectItem id="hsb">HSB</SelectItem>
      </Select>
      <For each={getColorChannels(space())} keyed={channel => channel}>{channel => <ColorSlider colorSpace={space()} channel={channel()} />}</For>
      <ColorSlider channel="alpha"/>
    </ColorPicker>);
}
```

### Color wheel

This example combines a [ColorWheel](ColorWheel) and [ColorArea](ColorArea) to build an HSB color picker.

```tsx nativeModule="ColorPicker/solid-example-3.tsx" solid="ColorPicker/solid-example-3.tsx" files={["packages/dev/s2-docs/pages/react-aria/ColorPicker/solid-example-3.tsx"]} 
import { ColorPicker } from './ColorPicker/vanilla/ColorPicker.solid';
import { ColorWheel } from './ColorPicker/vanilla/ColorWheel.solid';
import { ColorArea } from './ColorPicker/vanilla/ColorArea.solid';
function Example(props: Record<string, unknown>) {
    return (<ColorPicker label="Stroke color" defaultValue="#345">
  <ColorWheel />
  <ColorArea colorSpace="hsb" xChannel="saturation" yChannel="brightness" style={{
            "width": '100px',
            "height": '100px',
            "position": 'absolute',
            "top": 'calc(50% - 50px)',
            "left": 'calc(50% - 50px)'
        }}/>
    </ColorPicker>);
}
```

### Channel fields

This example uses [ColorField](ColorField) to allow a user to edit the value of each color channel as a number, along with a [Select](Select) to switch between color spaces.

```tsx nativeModule="ColorPicker/solid-example-4.tsx" solid="ColorPicker/solid-example-4.tsx" files={["packages/dev/s2-docs/pages/react-aria/ColorPicker/solid-example-4.tsx"]} 
import { getColorChannels, type ColorSpace } from 'solid-aria-components';
import { ColorPicker } from './ColorPicker/vanilla/ColorPicker.solid';
import { ColorArea } from './ColorPicker/vanilla/ColorArea.solid';
import { ColorSlider } from './ColorPicker/vanilla/ColorSlider.solid';
import { Select, SelectItem } from './ColorPicker/vanilla/Select.solid';
import { ColorField } from './ColorPicker/vanilla/ColorField.solid';
import { createSignal, For } from 'solid-js';
function Example() {
    let [space, setSpace] = createSignal<ColorSpace>('rgb');
    return (<ColorPicker label="Color" defaultValue="#f80">
      <ColorArea colorSpace="hsb" xChannel="saturation" yChannel="brightness"/>
      <ColorSlider colorSpace="hsb" channel="hue"/>
      <Select aria-label="Color space" selectedKey={space()} onSelectionChange={s => setSpace(s as ColorSpace)}>
        <SelectItem id="rgb">RGB</SelectItem>
        <SelectItem id="hsl">HSL</SelectItem>
        <SelectItem id="hsb">HSB</SelectItem>
      </Select>
      <div style={{ "display": 'flex', "gap": "4px", "width": "192px" }}>
        <For each={getColorChannels(space())} keyed={channel => channel}>{channel => <ColorField colorSpace={space()} channel={channel()} style={{ "flex": "1" }} />}</For>
      </div>
    </ColorPicker>);
}
```

### Swatches

This example uses a [ColorSwatchPicker](ColorSwatchPicker) to provide color presets for a color picker.

```tsx nativeModule="ColorPicker/solid-example-5.tsx" solid="ColorPicker/solid-example-5.tsx" files={["packages/dev/s2-docs/pages/react-aria/ColorPicker/solid-example-5.tsx"]} 
import { ColorPicker } from './ColorPicker/vanilla/ColorPicker.solid';
import { ColorArea } from './ColorPicker/vanilla/ColorArea.solid';
import { ColorSlider } from './ColorPicker/vanilla/ColorSlider.solid';
import { ColorSwatchPicker, ColorSwatchPickerItem } from './ColorPicker/vanilla/ColorSwatchPicker.solid';
function Example(props: Record<string, unknown>) {
    return (<ColorPicker label="Color" defaultValue="#A00">
  <ColorArea colorSpace="hsb" xChannel="saturation" yChannel="brightness"/>
  <ColorSlider colorSpace="hsb" channel="hue"/>
  <ColorSwatchPicker>
    <ColorSwatchPickerItem color="#A00"/>
    <ColorSwatchPickerItem color="#f80"/>
    <ColorSwatchPickerItem color="#080"/>
    <ColorSwatchPickerItem color="#08f"/>
    <ColorSwatchPickerItem color="#008"/>
  </ColorSwatchPicker>
    </ColorPicker>);
}
```
