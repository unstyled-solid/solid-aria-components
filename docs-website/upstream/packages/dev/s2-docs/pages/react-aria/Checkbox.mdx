## Selection

Use the `isSelected` or `defaultSelected` prop to set the selection state, and `onChange` to handle selection changes. The `isIndeterminate` prop overrides the selection state regardless of user interaction.

```tsx render solid="./Checkbox/solid-selection.tsx" wide docs={docs.exports.Checkbox} links={docs.links} props={['isIndeterminate']}
"use client";
import {Checkbox} from './Checkbox/vanilla/Checkbox.solid';
import {createSignal} from 'solid-js';

function Example(props) {
  let [selected, setSelection] = createSignal(false);

  return (
    <>
      <Checkbox
        {...props}
        ///- begin highlight -///
        isSelected={selected()}
        onChange={setSelection}
        /* PROPS */
        ///- end highlight -///
      >
        Subscribe
      </Checkbox>
      <p>{`You are ${props.isIndeterminate ? 'partially subscribed' : selected() ? 'subscribed' : 'unsubscribed'}`}</p>
    </>
  );
}
```

## Forms

Use the `name` and `value` props to submit the checkbox to the server. Set the `isRequired` prop to validate the user selects the checkbox, or implement custom client or server-side validation. See the [Forms](forms) guide to learn more.

```tsx render solid="./Checkbox/solid-forms.tsx"
"use client";
import {Checkbox} from './Checkbox/vanilla/Checkbox.solid';
import {Button} from './Button/vanilla/Button.solid';
import {Form} from './Button/vanilla/Form.solid';

<Form>
  <Checkbox
    ///- begin highlight -///
    name="terms"
    value="agree"
    isRequired
    description="By clicking this checkbox, you agree to the terms and conditions.">
    {/*- end highlight -*/}
    Accept terms and conditions
  </Checkbox>
  <Button type="submit" style={{'margin-top': '8px'}}>Submit</Button>
</Form>
```
