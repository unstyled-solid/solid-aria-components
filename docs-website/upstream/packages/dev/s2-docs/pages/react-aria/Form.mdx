## Submitting data

When using React 19, use the `action` prop to handle form submission. This receives a [FormData](https://developer.mozilla.org/en-US/docs/Web/API/FormData) object containing the values for each form field. In React 18 or earlier, use the `onSubmit` event instead. See the [Forms](forms) guide details about integrating with frameworks.

<ExampleSwitcher type={null} examples={['React 19', 'React 18']}>

```tsx nativeModule="Form/solid-example-3.tsx" solid="Form/solid-example-3.tsx" files={["packages/dev/s2-docs/pages/react-aria/Form/solid-example-3.tsx"]} 
import { Form } from './Form/vanilla/Form.solid';
import { TextField } from './Form/vanilla/TextField.solid';
import { Button } from './Form/vanilla/Button.solid';
function Example(props: Record<string, unknown>) {
    return (<Form 
    /*- begin highlight -*/
    onSubmit={event => {
            event.preventDefault();
            let formData = new FormData(event.currentTarget);
            let name = formData.get('name');
            alert(`Hello, ${name}!`);
            event.currentTarget.reset();
        }}>
  {/*- end highlight -*/}
  <TextField name="name" label="Name" placeholder="Enter your full name"/>
  <Button type="submit">Submit</Button>
    </Form>);
}
```

```tsx nativeModule="Form/solid-example-4.tsx" solid="Form/solid-example-4.tsx" files={["packages/dev/s2-docs/pages/react-aria/Form/solid-example-4.tsx"]} 
import { Form } from './Form/vanilla/Form.solid';
import { TextField } from './Form/vanilla/TextField.solid';
import { Button } from './Form/vanilla/Button.solid';
function Example(props: Record<string, unknown>) {
    return (<Form 
    /*- begin highlight -*/
    onSubmit={event => {
            // Prevent default browser page refresh.
            event.preventDefault();
            // Get data from form.
            let form = event.target as HTMLFormElement;
            let formData = new FormData(form);
            let name = formData.get('name');
            alert(`Hello, ${name}!`);
            // Reset form after submission.
            form.reset();
        }}>
  {/*- end highlight -*/}
  <TextField name="name" label="Name" placeholder="Enter your full name"/>
  <Button type="submit">Submit</Button>
    </Form>);
}

## Validation

Use validation props on each form field to prevent submission of invalid values. When `validationBehavior="aria"`, form submission will not be prevented and validation will occur as the value is edited instead of on form submission. See the [Forms](forms) guide for more details.

```tsx nativeModule="Form/solid-example-5.tsx" solid="Form/solid-example-5.tsx" files={["packages/dev/s2-docs/pages/react-aria/Form/vanilla/Form.solid.tsx","packages/dev/s2-docs/pages/react-aria/Form/vanilla/Form.css"]}  docs={docs.exports.Form} links={docs.links} props={['validationBehavior']} type="vanilla" 
import { Form } from './Form/vanilla/Form.solid';
import { TextField } from './Form/vanilla/TextField.solid';
import { Button } from './Form/vanilla/Button.solid';
function Example(props: Record<string, unknown>) {
    return (<Form {...props}>
  <TextField label="Username" placeholder="Choose a username" isRequired validate={value => value === 'admin' ? 'Nice try.' : null} name="username" defaultValue="admin"/>
  <Button type="submit">Submit</Button>
    </Form>);
}
```

### Server validation

Use the `validationErrors` prop to provide server validation errors to the fields within a `<Form>`.

```tsx nativeModule="Form/solid-example-6.tsx" solid="Form/solid-example-6.tsx" files={["packages/dev/s2-docs/pages/react-aria/Form/solid-example-6.tsx"]} 
import { Form } from './Form/vanilla/Form.solid';
import { TextField } from './Form/vanilla/TextField.solid';
function Example(props: Record<string, unknown>) {
    return (<Form 
    /*- begin highlight -*/
    validationErrors={{
            username: 'This username is not available.'
        }}>
  {/*- end highlight -*/}
  <TextField name="username" label="Username" placeholder="Choose a username" defaultValue="admin"/>
    </Form>);
}
```

### Focus management

By default, after a user submits a form with validation errors, the first invalid field will be focused. To prevent this, call `preventDefault()` in the `onInvalid` event. This example moves focus to an [alert](https://developer.mozilla.org/en-US/docs/Web/Accessibility/ARIA/Roles/alert_role) at the top of the form.

```tsx nativeModule="Form/solid-example-7.tsx" solid="Form/solid-example-7.tsx" files={["packages/dev/s2-docs/pages/react-aria/Form/solid-example-7.tsx"]} 
import { Form } from './Form/vanilla/Form.solid';
import { TextField } from './Form/vanilla/TextField.solid';
import { Button } from './Form/vanilla/Button.solid';
import { createSignal, onSettled } from 'solid-js';
function ErrorSummary() {
    let element: HTMLDivElement | undefined;
    onSettled(() => {element?.focus();});
    return <div role="alert" tabindex={-1} ref={node => {element = node;}}>
        <h3>Unable to submit</h3>
        <p>Please fix the validation errors below, and re-submit the form.</p>
    </div>;
}
function Example() {
    let [isInvalid, setInvalid] = createSignal(false);
    return (<Form 
    /*- begin highlight -*/
    onInvalid={e => {
            e.preventDefault();
            setInvalid(true);
        }} 
    /*- end highlight -*/
    onSubmit={e => {
            e.preventDefault();
            setInvalid(false);
        }} onReset={() => setInvalid(false)}>
      {isInvalid() && <ErrorSummary />}
      <TextField name="firstName" isRequired label="First Name" placeholder="Enter your first name"/>
      <TextField name="lastName" isRequired label="Last Name" placeholder="Enter your last name"/>
      <div style={{ "display": 'flex', "gap": "8px" }}>
        <Button type="submit">Submit</Button>
        <Button type="reset" variant="secondary">Reset</Button>
      </div>
    </Form>);
}
```
