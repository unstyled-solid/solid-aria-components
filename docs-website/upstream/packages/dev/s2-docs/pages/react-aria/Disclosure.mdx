## Expanding

Use the `isExpanded` or `defaultExpanded` prop to set the expanded state, and `onExpandedChange` to handle user interactions.

```tsx  nativeModule="Disclosure/solid-example-3.tsx" solid="Disclosure/solid-example-3.tsx"
import { Disclosure, DisclosureHeader, DisclosurePanel } from './Tabs/vanilla/Disclosure';
import { createSignal } from 'solid-js';
function Example(props: Record<string, unknown>) {
  let [isExpanded, setIsExpanded] = createSignal(true);
  return <Disclosure
  /*- begin highlight -*/ isExpanded={isExpanded()} onExpandedChange={setIsExpanded}>
      {/*- end highlight -*/}
      <DisclosureHeader>Download, Install, and Set Up</DisclosureHeader>
      <DisclosurePanel>Instructions on how to download, install, and set up</DisclosurePanel>
    </Disclosure>;
}
```

## Content

Add interactive elements such as buttons adjacent to the disclosure heading by wrapping them in a `<div>`. `<Heading>` and `<Button>` must not have interactive children.

```tsx  nativeModule="Disclosure/solid-example-4.tsx" solid="Disclosure/solid-example-4.tsx"
import { Disclosure, DisclosureHeader, DisclosurePanel } from './Tabs/vanilla/Disclosure';
import { Button } from './Tabs/vanilla/Button';
import { Settings } from './Tabs/icons';
function Example(props: Record<string, unknown>) {
  return <Disclosure style={{
    "width": "200px"
  }}>
  <div style={{
      "display": 'flex',
      "align-items": 'center',
      "gap": "8px"
    }}>
    {/*- begin highlight -*/}
    <DisclosureHeader style={{
        "flex": 1
      }}>Files</DisclosureHeader>
    <Button aria-label="Settings"><Settings /></Button>
    {/*- end highlight -*/}
  </div>
  <DisclosurePanel>Files content</DisclosurePanel>
</Disclosure>;
}
```
