## Value

Use the `value` or `defaultValue` prop to set the color value. The value may be a string or <TypeLink type={docs.exports.Color} links={docs.links} /> object, parsed using the <TypeLink type={docs.exports.parseColor} links={docs.links} /> function.

The `onChange` event is called as the user drags, and `onChangeEnd` is called when the thumb is released. These are always called with a `Color` object.

```tsx nativeModule="ColorWheel/solid-example-3.tsx" solid="ColorWheel/solid-example-3.tsx"  files={["packages/dev/s2-docs/pages/react-aria/ColorWheel/solid-example-3.tsx"]}
import {parseColor} from 'solid-aria-components';
import {ColorWheel} from './ColorWheel/vanilla/ColorWheel.solid';
import {createSignal} from 'solid-js';
function Example() {
  let [currentValue, setCurrentValue] = createSignal(parseColor('hsl(50, 100%, 50%)'));
  let [finalValue, setFinalValue] = createSignal(currentValue());

  return (
    <>
      <ColorWheel
        /*- begin highlight -*/
        value={currentValue()}
        onChange={setCurrentValue}
        onChangeEnd={setFinalValue} />
        {/*- end highlight -*/}
      <pre style={{'font-size': '12px'}}>
        onChange value: {currentValue().toString('hex')}{'\n'}
        onChangeEnd value: {finalValue().toString('hex')}
      </pre>
    </>
  );
}
```
